cmake_minimum_required(VERSION 3.25)
project(VintageDrummer VERSION 1.0.1 LANGUAGES C CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Supported build: Windows x64 with Visual Studio 2022/MSVC. See BUILDING.md.")
endif()
enable_language(RC)
set(IPLUG2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/iPlug2")
if(NOT EXISTS "${IPLUG2_DIR}/Dependencies/IPlug/VST3_SDK/pluginterfaces/base/funknown.h")
  message(FATAL_ERROR "Dependencies missing. Run: python scripts/bootstrap.py")
endif()
set(IPLUG_DEPLOY_PLUGINS OFF CACHE BOOL "Do not install into the user's DAW folders")
set(IGRAPHICS_BACKEND NANOVG CACHE STRING "Drawing backend")
set(IGRAPHICS_RENDERER GL2 CACHE STRING "Renderer")
include("${IPLUG2_DIR}/iPlug2.cmake")
find_package(iPlug2 REQUIRED)
add_subdirectory(vintage_drummer)
